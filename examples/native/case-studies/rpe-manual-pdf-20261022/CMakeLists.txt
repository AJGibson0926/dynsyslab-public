cmake_minimum_required(VERSION 3.24)
project(RpeManualPdfComparison LANGUAGES CXX)
if(NOT DYNSYSLAB_SDK_DIR)
  message(FATAL_ERROR "Set DYNSYSLAB_SDK_DIR to the installed DynSysLab root")
endif()
add_executable(rpe_manual_pdf experiment.cpp)
target_compile_features(rpe_manual_pdf PRIVATE cxx_std_17)
target_include_directories(rpe_manual_pdf PRIVATE "${DYNSYSLAB_SDK_DIR}/include")
foreach(module foundation eigenfunctions koopman)
  target_link_libraries(rpe_manual_pdf PRIVATE "${DYNSYSLAB_SDK_DIR}/lib/dynsyslab_${module}.lib")
endforeach()
if(DYNSYSLAB_SUITE_BUILD AND DYNSYSLAB_SOURCE_DIR)
  add_executable(rpe_manual_pdf_plots plots.cpp)
  target_compile_features(rpe_manual_pdf_plots PRIVATE cxx_std_17)
  target_compile_definitions(rpe_manual_pdf_plots PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_include_directories(rpe_manual_pdf_plots PRIVATE "${DYNSYSLAB_SOURCE_DIR}/viewer/native/include" "${DYNSYSLAB_SDK_DIR}/include")
  foreach(module viewer_services ui_support viewer_interaction)
    target_link_libraries(rpe_manual_pdf_plots PRIVATE "${DYNSYSLAB_SUITE_BUILD}/lib/dynsyslab_${module}.lib")
  endforeach()
  foreach(module foundation eigenfunctions koopman operations)
    target_link_libraries(rpe_manual_pdf_plots PRIVATE "${DYNSYSLAB_SDK_DIR}/lib/dynsyslab_${module}.lib")
  endforeach()
  target_link_libraries(rpe_manual_pdf_plots PRIVATE comctl32 gdiplus shell32 shlwapi dwmapi uxtheme ole32)
endif()
