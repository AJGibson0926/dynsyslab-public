cmake_minimum_required(VERSION 3.24)
project(KoopmanFeedbackBenchmarkPlots LANGUAGES CXX)
add_executable(koopman_feedback_plots plots.cpp)
target_compile_features(koopman_feedback_plots PRIVATE cxx_std_17)
target_compile_definitions(koopman_feedback_plots PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(koopman_feedback_plots PRIVATE
  "${DYNSYSLAB_SOURCE_DIR}/native/include" "${DYNSYSLAB_SOURCE_DIR}/viewer/native/include")
foreach(module viewer_services ui_support viewer_interaction foundation eigenfunctions koopman reconstruction operations)
  target_link_libraries(koopman_feedback_plots PRIVATE "${DYNSYSLAB_SUITE_BUILD}/lib/dynsyslab_${module}.lib")
endforeach()
target_link_libraries(koopman_feedback_plots PRIVATE comctl32 gdiplus shell32 shlwapi dwmapi uxtheme ole32)
