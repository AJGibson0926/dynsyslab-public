cmake_minimum_required(VERSION 3.24)
project(RpeRetrainedPdfComparison LANGUAGES CXX)
add_executable(rpe_retrained_pdf experiment.cpp)
target_compile_features(rpe_retrained_pdf PRIVATE cxx_std_17)
target_include_directories(rpe_retrained_pdf PRIVATE "${DYNSYSLAB_SDK_DIR}/include")
foreach(module foundation eigenfunctions koopman control)
  target_link_libraries(rpe_retrained_pdf PRIVATE "${DYNSYSLAB_SDK_DIR}/lib/dynsyslab_${module}.lib")
endforeach()
if(DYNSYSLAB_SUITE_BUILD AND DYNSYSLAB_SOURCE_DIR)

    target_link_libraries(rpe_retrained_plots PRIVATE "${DYNSYSLAB_SUITE_BUILD}/lib/dynsyslab_${module}.lib")

  endforeach()
