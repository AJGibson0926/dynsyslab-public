cmake_minimum_required(VERSION 3.24)
project(RpeRingdownFigure LANGUAGES CXX)
add_executable(ringdown_figure ringdown_figure.cpp)
target_compile_features(ringdown_figure PRIVATE cxx_std_17)
target_compile_definitions(ringdown_figure PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(ringdown_figure PRIVATE "${DYNSYSLAB_SOURCE_DIR}/native/include" "${DYNSYSLAB_SOURCE_DIR}/viewer/native/include")
foreach(module viewer_services ui_support viewer_interaction foundation eigenfunctions koopman reconstruction operations)
  target_link_libraries(ringdown_figure PRIVATE "${DYNSYSLAB_SUITE_BUILD}/lib/dynsyslab_${module}.lib")
endforeach()
target_link_libraries(ringdown_figure PRIVATE comctl32 gdiplus shell32 shlwapi dwmapi uxtheme ole32)
