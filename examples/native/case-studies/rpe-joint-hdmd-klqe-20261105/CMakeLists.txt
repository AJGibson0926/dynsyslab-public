cmake_minimum_required(VERSION 3.24)
project(RpeJointHdmdKlqe LANGUAGES CXX)
add_executable(rpe_joint_klqe experiment.cpp)
target_compile_features(rpe_joint_klqe PRIVATE cxx_std_17)
target_include_directories(rpe_joint_klqe PRIVATE "${DYNSYSLAB_SOURCE_DIR}/native/include")
foreach(module foundation eigenfunctions koopman spectral reconstruction estimation linear_analysis numerics)
  target_link_libraries(rpe_joint_klqe PRIVATE "${DYNSYSLAB_SUITE_BUILD}/lib/dynsyslab_${module}.lib")
endforeach()
add_executable(rpe_joint_klqe_plots plots.cpp)
target_compile_features(rpe_joint_klqe_plots PRIVATE cxx_std_17)
target_compile_definitions(rpe_joint_klqe_plots PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(rpe_joint_klqe_plots PRIVATE "${DYNSYSLAB_SOURCE_DIR}/native/include" "${DYNSYSLAB_SOURCE_DIR}/viewer/native/include")
foreach(module viewer_services ui_support viewer_interaction foundation eigenfunctions koopman reconstruction operations)
  target_link_libraries(rpe_joint_klqe_plots PRIVATE "${DYNSYSLAB_SUITE_BUILD}/lib/dynsyslab_${module}.lib")
endforeach()
target_link_libraries(rpe_joint_klqe_plots PRIVATE comctl32 gdiplus shell32 shlwapi dwmapi uxtheme ole32)
