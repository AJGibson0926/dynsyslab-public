cmake_minimum_required(VERSION 3.20)
project(MiscellaneousComputes LANGUAGES CXX)
option(MISC_BUILD_COMPANIONS "Also build the existing receiver and trajectory-resolution tools" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(Eigen3 CONFIG REQUIRED)
foreach(name foundation identification observation)
  find_library(${name}_library dynsyslab_${name} PATHS "${DYNSYSLAB_SUITE_BUILD}/lib" NO_DEFAULT_PATH REQUIRED)
endforeach()
add_library(misc_kernels SHARED hankel.cpp local_rhs.cpp rollout_kernel.cpp polynomial_kernel.cpp crossing_kernel.cpp delay_rollout_kernel.cpp)
set_target_properties(misc_kernels PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_link_libraries(misc_kernels PRIVATE Eigen3::Eigen)
add_executable(misc_computes cli.cpp rollout_cli.cpp crossing_cli.cpp)
target_include_directories(misc_computes PRIVATE "${ROOT}/native/include")
target_link_libraries(misc_computes PRIVATE misc_kernels "${foundation_library}" "${identification_library}" "${observation_library}")
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME miscellaneous_computes_regression COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/test_sidecar.py" $<TARGET_FILE:misc_computes> $<TARGET_FILE:misc_kernels>)
  add_test(NAME miscellaneous_computations COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/test_computations.py" $<TARGET_FILE:misc_kernels>)
  find_program(MISC_NATIVE_CLI NAMES dynsyslab-cli PATHS "${DYNSYSLAB_SUITE_BUILD}/bin" NO_DEFAULT_PATH)
  if(MISC_NATIVE_CLI)
    set(closure_cli "${MISC_NATIVE_CLI}")
  endif()
  add_test(NAME miscellaneous_toolkit_cli COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/test_toolkit_cli.py" $<TARGET_FILE:misc_computes> $<TARGET_FILE:misc_kernels> ${closure_cli})
  if(WIN32)
    set_tests_properties(miscellaneous_computes_regression miscellaneous_toolkit_cli PROPERTIES
      ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${DYNSYSLAB_SUITE_BUILD}/bin")
  endif()
endif()
if(MISC_BUILD_COMPANIONS)
  add_subdirectory(../receiver_reconstruction receiver)
  add_subdirectory(../trajectory_resolution resolution)
endif()
