cmake_minimum_required(VERSION 3.20)
project(ReceiverReconstructionSidecars LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(DYNSYSLAB_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_library(FOUNDATION dynsyslab_foundation PATHS "${DYNSYSLAB_SUITE_BUILD}/lib" NO_DEFAULT_PATH REQUIRED)
add_library(receiver_lookup STATIC lookup.cpp)
target_include_directories(receiver_lookup PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_library(receiver_records STATIC signals.cpp rings.cpp)
target_include_directories(receiver_records PUBLIC "${DYNSYSLAB_SOURCE_DIR}/native/include")
target_link_libraries(receiver_records PUBLIC "${FOUNDATION}")
add_executable(receiver_sidecar cli.cpp)
target_link_libraries(receiver_sidecar PRIVATE receiver_lookup receiver_records)
include(CTest)
if(BUILD_TESTING)
  add_executable(receiver_lookup_test lookup_test.cpp)
  target_link_libraries(receiver_lookup_test PRIVATE receiver_lookup)
  add_test(NAME receiver_lookup_test COMMAND receiver_lookup_test)
  add_executable(receiver_records_test records_test.cpp)
  target_link_libraries(receiver_records_test PRIVATE receiver_records)
  add_test(NAME receiver_records_test COMMAND receiver_records_test)
endif()
